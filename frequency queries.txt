import java.util.*;

public class Solution {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();

        Map<Integer, Integer> freq = new HashMap<>();
        Map<Integer, Integer> countFreq = new HashMap<>();

        for (int i = 0; i < q; i++) {
            int operation = sc.nextInt();
            int x = sc.nextInt();

            if (operation == 1 || operation == 2) {
                int oldFreq = freq.getOrDefault(x, 0);
                int newFreq = oldFreq;

                if (operation == 1) {
                    newFreq++;
                } else if (oldFreq > 0) {
                    newFreq--;
                }

                if (newFreq != oldFreq) {
                    if (oldFreq > 0) {
                        countFreq.put(oldFreq,
                                countFreq.get(oldFreq) - 1);
                    }

                    freq.put(x, newFreq);

                    if (newFreq > 0) {
                        countFreq.put(newFreq,
                                countFreq.getOrDefault(newFreq, 0) + 1);
                    }
                }
            } else if (operation == 3) {
                System.out.println(
                        countFreq.getOrDefault(x, 0) > 0 ? 1 : 0);
            }
        }

        sc.close();
    }
}
